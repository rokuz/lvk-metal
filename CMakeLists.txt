if(LVK_METAL_LVK_DIR STREQUAL LVK_METAL_BUNDLED_LVK_DIR)
  if(NOT EXISTS "${LVK_METAL_LVK_DIR}/CMakeLists.txt")
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    message(STATUS "lvk-metal: fetching dependencies via bootstrap.py")
    execute_process(COMMAND "${Python3_EXECUTABLE}" "${LVK_METAL_ROOT}/bootstrap.py" --use-unzip
                    WORKING_DIRECTORY "${LVK_METAL_ROOT}/3party" COMMAND_ERROR_IS_FATAL ANY)
  endif()
  set(LVK_WITH_SAMPLES OFF CACHE BOOL "" FORCE)
  set(LVK_WITH_TRACY OFF CACHE BOOL "" FORCE)
  set(LVK_WITH_IMPLOT OFF CACHE BOOL "" FORCE)
  set(LVK_WITH_SLANG OFF CACHE BOOL "" FORCE)
  set(LVK_DEPLOY_SCREENSHOT_TESTS OFF CACHE BOOL "" FORCE)
  set(LVK_WITH_GLFW ON CACHE BOOL "" FORCE)
  add_subdirectory("${LVK_METAL_LVK_DIR}" lightweightvk)

function(lvk_metal_target_add_common target)
  target_sources(${target} PRIVATE "${LVK_METAL_ROOT}/samples/common/app_common.cpp")
  target_include_directories(${target} PRIVATE "${LVK_METAL_ROOT}/samples/common" "${LVK_METAL_LVK_DIR}/third-party/deps/src/stb")
endfunction()

function(lvk_metal_target_add_upscaler target)
  target_sources(${target} PRIVATE "${LVK_METAL_ROOT}/samples/common/upscaler.cpp")
  target_include_directories(${target} PRIVATE "${LVK_METAL_ROOT}/samples/common")
  target_compile_definitions(${target} PRIVATE "LVK_METAL_CUNNY_PACKAGE=\"${LVK_METAL_ROOT}/samples/common/assets/cunny.mtlpackage\"")
endfunction()

function(lvk_metal_target_use_slang target shaders_dir)
  if(NOT LVK_METAL_WITH_SAMPLES_SLANG)
    return()
  endif()
  set(_slang_dir "${LVK_METAL_3PARTY_SRC}/slang")
  if(NOT EXISTS "${_slang_dir}/include/slang.h")
    message(FATAL_ERROR "Slang not found at ${_slang_dir}. Add it to 3party/bootstrap.json and re-run bootstrap.")
  endif()
  find_library(LVK_METAL_SLANG_LIB NAMES slang libslang PATHS "${_slang_dir}/lib" NO_DEFAULT_PATH REQUIRED)
  target_sources(${target} PRIVATE "${LVK_METAL_ROOT}/samples/common/slang_runtime.cpp")
  target_include_directories(${target} PRIVATE "${_slang_dir}/include" "${LVK_METAL_ROOT}/samples/common")
  target_link_libraries(${target} PRIVATE "${LVK_METAL_SLANG_LIB}")
  target_compile_definitions(${target} PRIVATE LVK_METAL_SAMPLE_USE_SLANG=1 "LVK_METAL_SAMPLE_SHADERS_DIR=\"${shaders_dir}\""
                                               "LVK_METAL_SAMPLE_COMMON_DIR=\"${LVK_METAL_ROOT}/samples/common\"")
  set_target_properties(${target} PROPERTIES BUILD_RPATH "${_slang_dir}/lib")
endfunction()

function(lvk_metal_build_bistro_deps)
  set(LVK_DEPS "${LVK_METAL_LVK_DIR}/third-party/deps/src")
  if(NOT TARGET meshoptimizer)
    add_subdirectory("${LVK_DEPS}/meshoptimizer" "${CMAKE_BINARY_DIR}/meshoptimizer")
  endif()
  if(NOT TARGET ktx)
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    set(KTX_FEATURE_TESTS OFF CACHE BOOL "" FORCE)
    set(KTX_FEATURE_TOOLS OFF CACHE BOOL "" FORCE)
    set(LIBKTX_VERSION_READ_ONLY OFF CACHE BOOL "")
    set(LIBKTX_VERSION_FULL ON CACHE BOOL "")
    set(LIBKTX_FEATURE_KTX1 ON CACHE BOOL "")
    set(LIBKTX_FEATURE_KTX2 ON CACHE BOOL "")
    set(LIBKTX_FEATURE_VK_UPLOAD OFF CACHE BOOL "")
    set(LIBKTX_FEATURE_GL_UPLOAD OFF CACHE BOOL "")
    set(LIBKTX_FEATURE_ETC_UNPACK ON CACHE BOOL "")
    add_subdirectory("${LVK_DEPS}/ktx-software/lib" "${CMAKE_BINARY_DIR}/ktx-software")
    if(CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
      if(TARGET basisu_encoder)
        target_compile_options(basisu_encoder PUBLIC "-Wno-deprecated-anon-enum-enum-conversion" "-Wno-deprecated-enum-enum-conversion")
      endif()
      if(TARGET ktx)
        target_compile_options(ktx PRIVATE "-Wno-deprecated-anon-enum-enum-conversion" "-Wno-deprecated-enum-enum-conversion")
      endif()
    endif()
  endif()
endfunction()
